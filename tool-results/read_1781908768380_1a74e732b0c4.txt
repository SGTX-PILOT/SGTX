   289→function InboxDrawer({ data, onClose, highPriority }: { data: DashboardData; onClose: () => void; highPriority: number }) {
   290→  const [aiSummary, setAiSummary] = useState<string | null>(null);
   291→  const [aiLoading, setAiLoading] = useState(false);
   292→  const [aiProvider, setAiProvider] = useState<string | null>(null);
   293→  const [hiddenIds, setHiddenIds] = useState<Set<string>>(new Set());
   294→  const [pendingId, setPendingId] = useState<string | null>(null);
   295→  const queryClient = useQueryClient();
   296→
   297→  const loadSummary = async () => {
   298→    if (aiLoading || aiSummary) return;
   299→    setAiLoading(true);
   300→    try {
   301→      const res = await fetch("/api/sgtx/ai/inbox-summary", {
   302→        method: "POST", headers: { "Content-Type": "application/json" },
   303→        body: JSON.stringify({ tenant: data.tenant.gtid }),
   304→      });
   305→      const d = await res.json();
   306→      setAiSummary(d.content);
   307→      setAiProvider(d.provider);
   308→    } catch { setAiSummary("AI summary unavailable."); }
   309→    finally { setAiLoading(false); }
   310→  };
   311→
   312→  // 3B.1.3.2 Smart Inbox item dismiss (after CTA click). Since we can't deep-link
   313→  // to arbitrary tabs safely, the CTA dismisses the item and surfaces a toast
   314→  // directing the user to the relevant portal tab.
   315→  const dismissItem = async (it: any) => {
   316→    if (pendingId) return;
   317→    setPendingId(it.id);
   318→    try {
   319→      const res = await fetch("/api/sgtx/inbox/dismiss", {
   320→        method: "POST", headers: { "Content-Type": "application/json" },
   321→        body: JSON.stringify({ inboxId: it.id }),
   322→      });
   323→      if (!res.ok) throw new Error("dismiss failed");
   324→      setHiddenIds((s) => new Set(s).add(it.id));
   325→      toast.success(it.ctaLabel ? `${it.ctaLabel} — done` : "Item dismissed", {
   326→        description: it.title,
   327→      });
   328→      // Refresh dashboard so the bell badge + recommended actions stay in sync
   329→      queryClient.invalidateQueries({ queryKey: ["dashboard"] });
   330→    } catch (e) {
   331→      toast.error("Could not dismiss item", { description: "Please try again." });
   332→    } finally {
   333→      setPendingId(null);
   334→    }
   335→  };
   336→
   337→  const snoozeItem = async (it: any, hours: number) => {
   338→    if (pendingId) return;
   339→    setPendingId(it.id);
   340→    try {
   341→      const res = await fetch("/api/sgtx/inbox/snooze", {
   342→        method: "POST", headers: { "Content-Type": "application/json" },
   343→        body: JSON.stringify({ inboxId: it.id, hours }),
   344→      });
   345→      if (!res.ok) throw new Error("snooze failed");
   346→      setHiddenIds((s) => new Set(s).add(it.id));
   347→      toast.success(`Snoozed for ${hours}h`, { description: it.title });
   348→      queryClient.invalidateQueries({ queryKey: ["dashboard"] });
   349→    } catch (e) {
   350→      toast.error("Could not snooze item", { description: "Please try again." });
   351→    } finally {
   352→      setPendingId(null);
   353→    }
   354→  };
   355→
   356→  const visibleInbox = (data.inbox || []).filter((it) => !hiddenIds.has(it.id));
   357→
   358→  return (
   359→    <>
   360→      <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={onClose} className="fixed inset-0 bg-black/50 z-40" />
   361→      <motion.div
   362→        initial={{ x: "100%" }} animate={{ x: 0 }} exit={{ x: "100%" }} transition={{ type: "spring", damping: 28 }}
   363→        className="fixed right-0 top-0 bottom-0 w-full sm:w-96 bg-card border-l border-border z-50 flex flex-col"
   364→      >
   365→        <div className="h-16 flex items-center justify-between px-5 border-b border-border">
   366→          <div>
   367→            <h3 className="font-semibold text-sm">Smart Inbox</h3>
   368→            <p className="text-[0.65rem] text-muted-foreground">{visibleInbox.length} actions · {highPriority} high priority</p>
   369→          </div>
   370→          <Button variant="ghost" size="icon" onClick={onClose} className="h-8 w-8"><X className="w-4 h-4" /></Button>
   371→        </div>
   372→
   373→        {/* AI Summary Card (Part 12A.1.3) */}
   374→        <div className="m-3 p-3 rounded-xl bg-gold/5 border border-gold/20">
   375→          <div className="flex items-center justify-between mb-2">
   376→            <p className="text-[0.6rem] tracking-widest text-gold uppercase font-semibold flex items-center gap-1"><Sparkles className="w-3 h-3" /> AI Summary</p>
   377→            {!aiSummary && !aiLoading && (
   378→              <button onClick={loadSummary} className="text-[0.65rem] text-gold hover:underline">Generate</button>
   379→            )}
   380→            {aiProvider && <span className="text-[0.55rem] text-muted-foreground">via {aiProvider}</span>}
   381→          </div>
   382→          {aiLoading ? (
   383→            <div className="flex items-center gap-2 text-xs text-muted-foreground"><Loader2 className="w-3 h-3 animate-spin" /> Analyzing your inbox…</div>
   384→          ) : aiSummary ? (
   385→            <p className="text-xs text-foreground/90 leading-relaxed">{aiSummary}</p>
   386→          ) : (
   387→            <p className="text-[0.65rem] text-muted-foreground">Click "Generate" for an AI plain-language summary of today's priorities. 🧠 A1 advisory.</p>
   388→          )}
   389→        </div>
   390→
   391→        {/* Recommended actions widget */}
   392→        {visibleInbox.length > 0 && (
   393→          <div className="mx-3 mb-2 p-3 rounded-xl bg-gold/10 border border-gold/30">
   394→            <p className="text-[0.6rem] tracking-widest text-gold uppercase font-semibold mb-2">📌 Recommended (1 click)</p>
   395→            <div className="space-y-2">
   396→              {visibleInbox.slice(0, 2).map((it) => (
   397→                <button key={it.id} onClick={() => dismissItem(it)} className="w-full text-left p-2 rounded-lg bg-background/60 hover:bg-background transition-colors">
   398→                  <p className="text-xs font-medium text-foreground line-clamp-1">{it.title}</p>
   399→                  <p className="text-[0.65rem] text-muted-foreground line-clamp-1">{it.description}</p>
   400→                </button>
   401→              ))}
   402→            </div>
   403→          </div>
   404→        )}
   405→
   406→        <ScrollArea className="flex-1 scroll-gold">
   407→          <div className="p-3 space-y-2">
   408→            {visibleInbox.length === 0 && (
   409→              <div className="p-6 text-center">
   410→                <p className="text-xs text-muted-foreground">🎉 All caught up. No pending actions.</p>
   411→              </div>
   412→            )}
   413→            {visibleInbox.map((it) => {
   414→              const color = it.priority >= 80 ? "#f87171" : it.priority >= 50 ? "#fbbf24" : "#60a5fa";
   415→              const isPending = pendingId === it.id;
   416→              return (
   417→                <div key={it.id} className="p-3 rounded-xl border border-border bg-background/40 hover:border-gold/40 transition-colors">
   418→                  <div className="flex items-start gap-2">

   462→function AssistantDrawer({ onClose, tenant }: { onClose: () => void; tenant: any }) {
   463→  const [messages, setMessages] = useState<{ role: "user" | "ai"; content: string; provider?: string }[]>(
   464→    tenant ? [{ role: "ai", content: `Hello, ${tenant.legalName}. I'm your SGTX AI Operations Assistant & Customer Care Chatbot (A1 advisory, z-ai glm-4-plus). I can answer questions, perform actions on your behalf (with PIN-based impersonation), and escalate to human support via VoIP callback. I never recommend counterparties — SGTX is a non-marketplace system.` }] : []
   465→  );
   466→  const [input, setInput] = useState("");
   467→  const [loading, setLoading] = useState(false);
   468→  const [showEscalate, setShowEscalate] = useState(false);
   469→  const [pinPrompt, setPinPrompt] = useState<string | null>(null);
   470→  const [pinInput, setPinInput] = useState("");
   471→  const scrollRef = useRef<HTMLDivElement>(null);
   472→
   473→  const send = async (text?: string) => {
   474→    const msg = (text ?? input).trim();
   475→    if (!msg || loading || !tenant) return;
   476→    setInput("");
   477→
   478→    // Handle "Talk to a human" escalation (3B.1.3.6)
   479→    if (msg.toLowerCase().includes("talk to a human") || msg.toLowerCase().includes("escalate")) {
   480→      setMessages((m) => [...m, { role: "user", content: msg }, { role: "ai", content: "I'll escalate this to a human support agent. They will have access to our conversation transcript and trade context. Would you like a VoIP callback? Click below to confirm your phone number and request a callback." }]);
   481→      setShowEscalate(true);
   482→      return;
   483→    }
   484→
   485→    setMessages((m) => [...m, { role: "user", content: msg }]);
   486→    setLoading(true);
   487→    try {
   488→      const res = await fetch("/api/sgtx/ai/chat", {
   489→        method: "POST", headers: { "Content-Type": "application/json" },
   490→        body: JSON.stringify({ tenant: tenant.gtid, message: msg }),
   491→      });
   492→      const data = await res.json();
   493→      setMessages((m) => [...m, { role: "ai", content: data.content, provider: data.provider }]);
   494→    } catch {
   495→      setMessages((m) => [...m, { role: "ai", content: "Sorry, I couldn't process that. Please try again." }]);
   496→    } finally {
   497→      setLoading(false);
   498→    }

   504→  };
   505→

   507→    setMessages((m) => [...m, { role: "ai", content: `PIN verified. I now have a short-lived, scope-limited JWT to perform: ${pinPrompt}. All impersonated actions are logged as Governor decisions (decision_type = 'customer_care_bot_impersonation').` }]);

   518→      <motion.div
   519→        initial={{ x: "100%" }} animate={{ x: 0 }} exit={{ x: "100%" }} transition={{ type: "spring", damping: 28 }}

   521→      >
   522→        <div className="h-16 flex items-center justify-between px-5 border-b border-border">
   523→          <div className="flex items-center gap-2">

   528→            </div>

   531→        </div>
