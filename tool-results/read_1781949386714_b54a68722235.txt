   411→function InboxDrawer({ data, onClose, highPriority }: { data: DashboardData; onClose: () => void; highPriority: number }) {
   412→  const [aiSummary, setAiSummary] = useState<string | null>(null);
   413→  const [aiLoading, setAiLoading] = useState(false);
   414→  const [aiProvider, setAiProvider] = useState<string | null>(null);
   415→  const [hiddenIds, setHiddenIds] = useState<Set<string>>(new Set());
   416→  const [pendingId, setPendingId] = useState<string | null>(null);
   417→  const queryClient = useQueryClient();
   418→
   419→  const loadSummary = async () => {
   420→    if (aiLoading || aiSummary) return;
   421→    setAiLoading(true);
   422→    try {
   423→      const res = await fetch("/api/sgtx/ai/inbox-summary", {
   424→        method: "POST", headers: { "Content-Type": "application/json" },
   425→        body: JSON.stringify({ tenant: data.tenant.gtid }),
   426→      });
   427→      const d = await res.json();
   428→      setAiSummary(d.content);
   429→      setAiProvider(d.provider);

   584→function AssistantDrawer({ onClose, tenant }: { onClose: () => void; tenant: any }) {
