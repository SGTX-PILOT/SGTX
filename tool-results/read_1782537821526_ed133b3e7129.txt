    12→    12→// ============ 10.2: File Dispute ============
    13→    13→export async function fileDispute(input: {
    14→    14→  ustn: string; tradeId?: string; filedByGtid: string; category: string;
    15→    15→  description: string; claimAmountUsd?: number; remedySought?: string;
    16→    16→  affectedPortion?: string; uploadedEvidence?: string[];
    17→    17→}): Promise<{ ok: true; disputeId: string } | { ok: false; reason: string; code?: string }> {
    18→    18→  const trade = await db.trade.findUnique({ where: { ustn: input.ustn } });
    19→    19→  if (!trade) return { ok: false, code: "G10U1_NOT_FOUND", reason: "Trade not found." };
    20→    20→  if (input.description.trim().length < 10) return { ok: false, code: "G10U1_DESC", reason: "Description must be ≥10 chars." };
    21→    21→
    22→    22→  const dispute = await db.dispute.create({
    23→    23→    data: { tradeId: trade.id, type: input.category, status: "FILED",
    24→    24→      filedByGtid: input.filedByGtid, claimAmountUsd: input.claimAmountUsd || 0,
    25→    25→      description: input.description, evidenceCount: (input.uploadedEvidence || []).length },
    26→    26→  });
    27→    27→  await db.trade.update({ where: { id: trade.id }, data: { status: "DISPUTED" } });
    28→    28→  // Freeze pending settlement instructions (if model exists)
    29→    29→  try {
    30→    30→    await (db as any).settlementInstruction?.updateMany({
    31→    31→      where: { ustn: input.ustn, status: { in: ["PENDING_APPROVAL", "APPROVED"] } },
    32→    32→      data: { status: "FROZEN", frozenReason: `Dispute ${dispute.id} filed` },
    33→    33→    });
    34→    34→  } catch { /* settlementInstruction model may not exist */ }
    35→    35→  // Notify counterparty
    36→    36→  const counterparty = trade.buyerGtid === input.filedByGtid ? trade.sellerGtid : trade.buyerGtid;
    37→    37→  await db.inboxItem.create({ data: { tenantGtid: counterparty, tradeId: trade.id,
    38→    38→    category: "COMPLIANCE", priority: 95,
    39→    39→    title: `Dispute filed — ${dispute.id.slice(-8)} (${input.category})`,
    40→    40→    description: `${input.description.slice(0, 100)}… FeeLock frozen.`, ctaLabel: "Open Mediation" }});
    41→    41→  // Auto-trigger evidence + triage
    42→    42→  await compileEvidence(dispute.id);
    43→    43→  await runDisputeTriage(dispute.id);
    44→    44→  return { ok: true, disputeId: dispute.id };
    45→    45→}

    47→    47→// ============ 10.3: Evidence Autocompiler ============
    48→    48→export async function compileEvidence(disputeId: string): Promise<{ ok: true; evidenceId: string; packageHash: string; verificationToken: string } | { ok: false; reason: string }> {
    49→    49→  const dispute = await db.dispute.findUnique({ where: { id: disputeId }, include: { trade: { include: { documents: true, shipments: true, activities: true, invoices: true, labTests: true, qcInspections: true, chatMessages: true } } } });
    50→    50→  if (!dispute) return { ok: false, reason: "Dispute not found." };
    51→    51→  const contents: string[] = [`Trade: ${dispute.trade.ustn}`, `Commodity: ${dispute.trade.commodity}`, `Incoterm: ${dispute.trade.incoterm}`, `Value: $${dispute.trade.tradeValueUsd}`];
    52→    52→  for (const s of dispute.trade.shipments) contents.push(`Shipment ${s.containerNo}: status ${s.status}`);
    53→    53→  for (const qc of dispute.trade.qcInspections) contents.push(`QC: ${qc.result} (${qc.defectCount} defects)`);
    54→    54→  for (const lab of dispute.trade.labTests) contents.push(`Lab: ${lab.testType} — ${lab.passFail}`);
    55→    55→  for (const doc of dispute.trade.documents) contents.push(`Doc: ${doc.type} — ${doc.title} — ${doc.status}`);
    56→    56→  contents.push(`Messages: ${dispute.trade.chatMessages.length}`);
    57→    57→  const contentsJson = JSON.stringify(contents);
    58→    58→  const packageHash = "sha256:" + crypto.createHash("sha256").update(contentsJson).digest("hex");
    59→    59→  const loomHash = "sha256:loom:" + crypto.createHash("sha256").update(packageHash + disputeId).digest("hex").slice(0, 32);
    60→    60→  const verificationToken = "evd-" + crypto.randomBytes(8).toString("hex");
    61→    61→  const existing = await db.disputeEvidence.findUnique({ where: { disputeId } });
    62→    62→  if (existing) { await db.disputeEvidence.update({ where: { id: existing.id }, data: { packageHash, loomHash, contents: contentsJson, verificationToken } });
    63→    63→    return { ok: true, evidenceId: existing.id, packageHash, loomHash, verificationToken }; }
    64→    64→  const evidence = await db.disputeEvidence.create({ data: { disputeId, packageHash, loomHash, contents: contentsJson, verificationToken } });
    65→    65→  return { ok: true, evidenceId: evidence.id, packageHash, loomHash, verificationToken };
    66→    66→}

    69→    69→export async function runCausalAnalysis(disputeId: string): Promise<{ ok: true; rootCauses: any[]; summary: string } | { ok: false; reason: string }> {
    70→    70→  const dispute = await db.dispute.findUnique({ where: { id: disputeId }, include: { trade: true } });
    71→    71→  if (!dispute) return { ok: false, reason: "Dispute not found." };
    72→    72→  const rootCauses = dispute.type === "DELAY" ? [
    73→    73→    { factor: "port_strike", contribution: 0.54, description: "Port closure added 54% of total delay." },
    74→    74→    { factor: "carrier_rerouting", contribution: 0.32, description: "Carrier rerouted due to weather, adding 32%." },
    75→    75→    { factor: "customs_hold", contribution: 0.14, description: "Missing certificate caused 14% delay." },
    76→    76→  ] : dispute.type === "QUALITY" ? [
    77→    77→    { factor: "post_delivery_mishandling", contribution: 0.62, description: "Temperature log shows no deviation — damage likely post-delivery (62%)." },
    78→    78→    { factor: "preexisting_quality", contribution: 0.28, description: "Photos suggest pre-existing quality issue (28%)." },
    79→    79→    { factor: "transit_excursion", contribution: 0.10, description: "Minor transit excursion possible (10%)." },
    80→    80→  ] : [{ factor: "contract_breach", contribution: 0.70, description: "Primary breach (70%)." }, { factor: "communication_failure", contribution: 0.30, description: "Communication breakdown (30%)." }];
    81→    81→  const summary = `Caused primarily by ${rootCauses[0].factor.replace(/_/g, " ")} (${Math.round(rootCauses[0].contribution * 100)}%).`;
    82→    82→  await db.dispute.update({ where: { id: disputeId }, data: { aiRootCause: summary } });
    83→    83→  return { ok: true, rootCauses, summary };

    88→    88→  disputeId: string; senderGtid: string; senderName: string; senderRole: string;
    89→    89→  messageType: string; messageText?: string; offerAmountUsd?: number; offerConditions?: string[]; language?: string;
    90→    90→}): Promise<{ ok: true; messageId: string; sentimentFlag?: string } | { ok: false; reason: string }> {
    91→    91→  const signature = "zitadel:" + crypto.createHash("sha256").update(input.senderGtid + input.disputeId + Date.now()).digest("hex").slice(0, 32);
    92→    92→  let sentimentScore = 0, sentimentFlag = "neutral";
    93→    93→  if (input.messageText) { const t = input.messageText.toLowerCase();
    94→    94→    if (t.includes("unacceptable") || t.includes("reject")) { sentimentScore = -0.7; sentimentFlag = "hostile"; }

JSON.stringify(input.offerConditions) : null, language: input.language || "en", sentimentScore, sentimentFlag, signature } });
    97→    97→  return { ok: true, messageId: msg.id, sentimentFlag };
    98→    98→}

   118→   118→  await db.settlementProposal.update({ where: { id: proposal.id }, data: updates });
   119→   119→  const updated = await db.settlementProposal.findUnique({ where: { id: proposal.id } });

   122→   122→    await db.dispute.update({ where: { id: proposal.disputeId }, data: { status: "RESOLVED", resolution: `Settlement: $${proposal.amountUsd}` } });
   123→   123→    return { ok: true, bothAccepted: true, addendumSigned: true }; }
   124→   124→  return { ok: true, bothAccepted: false, addendumSigned: false };

   182→   182→// ============ 10.12: TRI Calculation (real DB metrics — Part 10 gap-fix) ============
   183→   183→//
   184→   184→// Previously the component scores were derived from `Math.random()` calls.
   185→   185→// Part 10 gap-fix replaces them with real queries against PaymentAttempt,
   186→   186→// SuspiciousActivityReport, Jurisdiction, Document, FinancingRepayment,
   187→   187→// FinancingRequest, DeFiPosition, and Dispute. Confidence is now derived
   188→   188→// from trade volume, history length, jurisdiction spread, and financier count.
   189→   189→export async function calculateTri(tenantGtid: string): Promise<{ triScore: number; confidence: number; components: any; status: string }> {
   190→   190→  const tenant = await db.tenant.findUnique({ where: { gtid: tenantGtid } });
   191→   191→  if (!tenant) throw new Error("Tenant not found");
   192→   192→
   193→   193→  const trades = await db.trade.findMany({
   194→   194→    where: { OR: [{ buyerGtid: tenantGtid }, { sellerGtid: tenantGtid }] },
   195→   195→    include: { documents: true },
   196→   196→  });
   197→   197→  const tradeUstns = trades.map((t) => t.ustn);
   198→   198→
   199→   199→  // ---------- settlementReliability ----------
   200→   200→  // (on_time_pct × 8) + max(0, 500 − avg_delay_days × 20). Default 500 when no payments.
   201→   201→  let settlementReliability: number;
   202→   202→  const paymentAttempts =
   203→   203→    tradeUstns.length > 0
   204→   204→      ? await db.paymentAttempt.findMany({ where: { ustn: { in: tradeUstns } } })
   205→   205→      : [];
   206→   206→  if (paymentAttempts.length === 0) {
   207→   207→    settlementReliability = 500;
   208→   208→  } else {
   209→   209→    const completed = paymentAttempts.filter((p) => p.status === "COMPLETED");
   210→   210→    const onTimePct = (completed.length / paymentAttempts.length) * 100;
   211→   211→    const avgDelayDays =
   212→   212→      completed.length > 0
   213→   213→        ? completed.reduce((s, p) => {
   214→   214→            const delay =
   215→   215→              p.completedAt && p.attemptedAt
   216→   216→                ? (p.completedAt.getTime() - p.attemptedAt.getTime()) / 86400000
   217→   217→                : 0;
   218→   218→            return s + Math.max(0, delay);
   219→   219→          }, 0) / completed.length
   220→   220→        : 0;
   221→   221→    settlementReliability = Math.min(
   222→   222→      1000,
   223→   223→      onTimePct * 8 + Math.max(0, 500 - avgDelayDays * 20),
   224→   224→    );
   225→   225→  }
   226→   226→
   227→   227→  // ---------- complianceHealth ----------
   228→   228→  // Start at 1000. Subtract: sanctionsCleared=false → -200, kybTier<2 → -200,
   229→   229→  // SAR count × 10 (up to 300), disputes in RESTRICTED jurisdictions → -50 each.
   230→   230→  let complianceHealth = 1000;
   231→   231→  if (!tenant.sanctionsCleared) complianceHealth -= 200;
   232→   232→  if (tenant.kybTier < 2) complianceHealth -= 200;
   233→   233→
   234→   234→  const tenantSars = await db.suspiciousActivityReport.count({
   235→   235→    where: { parties: { contains: tenantGtid } },
   236→   236→  });
   237→   237→  complianceHealth -= Math.min(300, tenantSars * 10);
   238→   238→
   239→   239→  const restrictedJurisdictions = await db.jurisdiction.findMany({
   240→   240→    where: { tier: { in: ["RESTRICTED", "BLOCKED"] } },
   241→   241→  });
   242→   242→  const restrictedCodes = new Set(restrictedJurisdictions.map((j) => j.countryCode));
   243→   243→  const disputes = await db.dispute.findMany({
   244→   244→    where: { trade: { OR: [{ buyerGtid: tenantGtid }, { sellerGtid: tenantGtid }] } },
   245→   245→    include: { trade: true },
   246→   246→  });
   247→   247→  let restrictedDisputeCount = 0;
   248→   248→  for (const d of disputes) {
   249→   249→    if (
   250→   250→      restrictedCodes.has(d.trade.originCountry) ||
   251→   251→      restrictedCodes.has(d.trade.destCountry)
   252→   252→    ) {
   253→   253→      restrictedDisputeCount++;
   254→   254→    }
   255→   255→  }
   256→   256→  complianceHealth -= restrictedDisputeCount * 50;
   257→   257→  complianceHealth = Math.max(0, complianceHealth);
   258→   258→
   259→   259→  // ---------- documentationQuality ----------
   260→   260→  // First-time acceptance rate = VERIFIED / (VERIFIED + REJECTED). Score = rate × 1000.
   261→   261→  // Default 850 when no docs.
   262→   262→  const allDocs = trades.flatMap((t) => t.documents);
   263→   263→  const verifiedDocs = allDocs.filter((d) => d.status === "VERIFIED").length;
   264→   264→  const rejectedDocs = allDocs.filter((d) => d.status === "REJECTED").length;
   265→   265→  const documentationQuality =
   266→   266→    verifiedDocs + rejectedDocs === 0
   267→   267→      ? 850
   268→   268→      : Math.round((verifiedDocs / (verifiedDocs + rejectedDocs)) * 1000);
   269→   269→
   270→   270→  // ---------- financingPerformance ----------
   271→   271→  // Score = 1000 − (defaults × 300) − (late_rate × 500). Default 900 when no financing.
   272→   272→  const financingRequests = await db.financingRequest.findMany({
   273→   273→    where: { borrowerGtid: tenantGtid },
   274→   274→    include: { repayments: true },
   275→   275→  });
   276→   276→  let defaults = 0;
   277→   277→  let totalRepayments = 0;
   278→   278→  let lateRepayments = 0;
   279→   279→  for (const req of financingRequests) {
   280→   280→    if (req.status === "REJECTED") defaults++;
   281→   281→    totalRepayments += req.repayments.length;
   282→   282→    const expectedDate = new Date(
   283→   283→      req.createdAt.getTime() + req.tenorDays * 86400000,
   284→   284→    );
   285→   285→    for (const rp of req.repayments) {
   286→   286→      if (rp.status !== "CONFIRMED") continue;
   287→   287→      if (rp.repaidAt && rp.repaidAt > expectedDate) lateRepayments++;
   288→   288→    }
   289→   289→  }
   290→   290→  // DeFi liquidations count as defaults.
   291→   291→  try {
   292→   292→    const defiDefaults = await db.deFiPosition.count({
   293→   293→      where: { borrowerGtid: tenantGtid, status: "LIQUIDATED" },
   294→   294→    });
   295→   295→    defaults += defiDefaults;
   296→   296→  } catch {
   297→   297→    /* DeFiPosition table may be empty/missing — ignore */
   298→   298→  }
   299→   299→  const lateRate = totalRepayments > 0 ? lateRepayments / totalRepayments : 0;
   300→   300→  const financingPerformance =
   301→   301→    financingRequests.length === 0
   302→   302→      ? 900
   303→   303→      : Math.max(0, 1000 - defaults * 300 - Math.round(lateRate * 500));
   304→   304→
   305→   305→  // ---------- disputeResolution ----------
   306→   306→  // no_arbitration_rate = disputes resolved without arbitration / total.
   307→   307→  // avg_resolution_days = avg(updatedAt − createdAt) for RESOLVED disputes.
   308→   308→  // Score = (no_arb_rate × 5) + 400 + max(0, 500 − avg_days × 5). Default 900.
   309→   309→  let disputeResolution: number;
   310→   310→  if (disputes.length === 0) {
   311→   311→    disputeResolution = 900;
   312→   312→  } else {
   313→   313→    const noArbitrationCount = disputes.filter(
   314→   314→      (d) => !["ARBITRATION", "ESCALATED"].includes(d.status),
   315→   315→    ).length;
   316→   316→    const noArbitrationRate = noArbitrationCount / disputes.length;
   317→   317→    const resolvedDisputes = disputes.filter((d) => d.status === "RESOLVED");
   318→   318→    const avgResolutionDays =
   319→   319→      resolvedDisputes.length > 0
   320→   320→        ? resolvedDisputes.reduce(
   321→   321→            (s, d) =>
   322→   322→              s + (d.updatedAt.getTime() - d.createdAt.getTime()) / 86400000,
   323→   323→            0,
   324→   324→          ) / resolvedDisputes.length
   325→   325→        : 0;
   326→   326→    disputeResolution = Math.min(
   327→   327→      1000,
   328→   328→      noArbitrationRate * 5 + 400 + Math.max(0, 500 - avgResolutionDays * 5),
   329→   329→    );
   330→   330→  }
   331→   331→
   332→   332→  // ---------- TRI Score (weighted) ----------
   333→   333→  const triScore = Math.round(
   334→   334→    settlementReliability * 0.25 +
   335→   335→      complianceHealth * 0.2 +
   336→   336→      documentationQuality * 0.15 +
   337→   337→      financingPerformance * 0.2 +
   338→   338→      disputeResolution * 0.2,
   339→   339→  );
   340→   340→
   341→   341→  // ---------- Confidence ----------
   342→   342→  // (√trade_count × 5) + (√(total_volume/10000) × 3) +
   343→   343→  // min(history_months/36 × 15, 15) + (jurisdiction_count × 2) + (financier_count × 1).
   344→   344→  // Capped at 100.
   345→   345→  const tradeCount = trades.length;
   346→   346→  const totalVolume = trades.reduce((s, t) => s + t.tradeValueUsd, 0);
   347→   347→  const historyMonths = tenant.createdAt
   348→   348→    ? Math.max(1, (Date.now() - tenant.createdAt.getTime()) / (30 * 86400000))
   349→   349→    : 1;
   350→   350→  const jurisdictionCount = new Set([
   351→   351→    ...trades.map((t) => t.originCountry),
   352→   352→    ...trades.map((t) => t.destCountry),
   353→   353→  ]).size;
   354→   354→  let financierCount = 0;
   355→   355→  try {
   356→   356→    const bids = await db.financingBid.findMany({
   357→   357→      where: { request: { borrowerGtid: tenantGtid } },
   358→   358→      select: { financierGtid: true },
   359→   359→    });
   360→   360→    financierCount = new Set(bids.map((b) => b.financierGtid)).size;
   361→   361→  } catch {
   362→   362→    /* ignore */
   363→   363→  }
   364→   364→
   365→   365→  const confidence = Math.min(
   366→   366→    100,
   367→   367→    Math.sqrt(tradeCount) * 5 +
   368→   368→      Math.sqrt(totalVolume / 10000) * 3 +
   369→   369→      Math.min((historyMonths / 36) * 15, 15) +
   370→   370→      jurisdictionCount * 2 +
   371→   371→      financierCount * 1,
   372→   372→  );
   373→   373→
   374→   374→  const status =
   375→   375→    triScore >= 900
   376→   376→      ? "Premier Trusted"
   377→   377→      : triScore >= 800
   378→   378→        ? "Advanced Trusted"
   379→   379→        : triScore >= 700
   380→   380→          ? "Trusted"
   381→   381→          : triScore >= 600
   382→   382→            ? "Verified"
   383→   383→            : triScore >= 500
   384→   384→              ? "Developing"
   385→   385→              : "Limited History";
   386→   386→  const components = {
   387→   387→    settlementReliability: Math.round(settlementReliability),
   388→   388→    complianceHealth,
   389→   389→    documentationQuality,
   390→   390→    financingPerformance,
   391→   391→    disputeResolution: Math.round(disputeResolution),
   392→   392→  };
   393→   393→  await db.triHistory.create({
   394→   394→    data: {
   395→   395→      tenantGtid,
   396→   396→      triScore,
   397→   397→      confidence,
   398→   398→      componentScores: JSON.stringify(components),
   399→   399→    },
   400→   400→  });
   401→   401→  return { triScore, confidence, components, status };
   402→   402→}
