    68→export async function signDocument(params: {
    69→  ustn?: string;
    70→  signerGtid: string;
    71→  signerName: string;
    72→  documentHash: string;
    73→  tradeValueUsd?: number;
    74→  forceType?: SignatureType;
    75→  documentType?: string;
    76→  hybridMode?: string; // 1.13.7 fallback
    77→}): Promise<{ id: string; type: SignatureType; provider: string; signatureValue: string; legalEffect: string }> {
    78→  const type = params.forceType || determineSignatureType(params.tradeValueUsd || 0);
    79→  const config = SIGNATURE_HIERARCHY[type];
    80→  const sigValue = type === "QES" ? "egyptTrust:" : type === "AES" ? "ed25519:" : "webauthn:";
    81→  const sig = sigValue + createHash("sha256").update(params.documentHash + params.signerGtid + Date.now()).digest("hex").slice(0, 64);
    82→
    83→  const record = await db.qesSignature.create({
    84→    data: {
    85→      ustn: params.ustn || null,
    86→      signerGtid: params.signerGtid,
    87→      signerName: params.signerName,
    88→      signatureType: type,
    89→      legalEffect: config.legalEffect,
    90→      provider: config.provider.split(" ")[0],
    91→      documentHash: params.documentHash,
    92→      signatureValue: sig,
    93→      documentType: params.documentType || null,
    94→      hybridMode: params.hybridMode || null,
    95→    },
    96→  });
    97→  return { id: record.id, type, provider: config.provider, signatureValue: sig, legalEffect: config.legalEffect };
    98→}

   101→export async function initiateQesRequest(params: {
   102→  documentSha256: string;
   103→  documentType: string;
   104→  ustn?: string;
   105→  signerGtid: string;
   106→  signerTsp: string; // EGYPT_TRUST | MISR
   107→  callbackUrl?: string;
   108→}): Promise<{ requestId: string; tspRequestUrl: string; expiresAt: Date; status: string }> {
   109→  const requestId = "QES-" + new Date().toISOString().slice(0, 10).replace(/-/g, "") + "-" + Math.random().toString(36).slice(2, 6).toUpperCase();
   110→  const expiresAt = new Date(Date.now() + 2 * 3600 * 1000); // 2-hour expiry
   111→  const tspRequestUrl = `https://ts.${params.signerTsp.toLowerCase().replace("_", "")}.com.eg/sign/${requestId}`;
   112→
   113→  await db.qesRequest.create({
   114→    data: {
   115→      requestId,
   116→      documentSha256: params.documentSha256,
   117→      documentType: params.documentType,
   118→      ustn: params.ustn || null,
   119→      signerGtid: params.signerGtid,
   120→      signerTsp: params.signerTsp,
   121→      status: "PENDING",
   122→      tspRequestUrl,
   123→      callbackUrl: params.callbackUrl || null,
   124→      expiresAt,
   125→    },
   126→  });
   127→  return { requestId, tspRequestUrl, expiresAt, status: "PENDING" };
   128→}

   179→export async function registerDevice(params: {
   180→  tenantGtid: string;
   181→  deviceFingerprint: string;
   182→  deviceName: string;
   183→  platform: string;
   184→  lastSeenIp?: string;
   185→  lastSeenCountry?: string;
   186→}): Promise<{ id: string; state: DeviceState; riskScore: number }> {
   187→  // Session risk engine — check for anomalies
   188→  let riskScore = 0;
   189→  const anomalies: SessionRiskType[] = [];
   190→
   191→  // Simulated anomaly detection (A2)
   192→  if (params.lastSeenCountry && params.lastSeenCountry === "TOR") { anomalies.push("tor_detected"); riskScore += 40; }
   193→  if (params.lastSeenIp && params.lastSeenIp.startsWith("10.")) { /* VPN heuristic */ }
   194→
   195→  const state: DeviceState = riskScore >= 70 ? "BLOCKED" : riskScore >= 40 ? "ELEVATED_RISK" : "NEW";
   196→
   197→  const device = await db.deviceTrust.create({
   198→    data: {
   199→      tenantGtid: params.tenantGtid,
   200→      deviceFingerprint: params.deviceFingerprint,
   201→      deviceName: params.deviceName,
   202→      platform: params.platform,
   203→      state,
   204→      lastSeenIp: params.lastSeenIp || null,
   205→      lastSeenCountry: params.lastSeenCountry || null,
   206→      riskScore,
   207→    },
   208→  });
   209→
   210→  // Log risk events
   211→  for (const anomaly of anomalies) {
   212→    await db.sessionRiskEvent.create({
   213→      data: {
   214→        tenantGtid: params.tenantGtid,
   215→        deviceFingerprint: params.deviceFingerprint,
   216→        eventType: anomaly,
   217→        severity: riskScore >= 70 ? "critical" : riskScore >= 40 ? "high" : "medium",
   218→        description: `Detected ${anomaly.replace(/_/g, " ")} during device registration`,
   219→        ipAddress: params.lastSeenIp || null,
   220→        countryCode: params.lastSeenCountry || null,
   221→      },
   222→    });
   223→  }
   224→
   225→  return { id: device.id, state, riskScore };
   226→}

   228→export async function performStepUpAuth(params: {
   229→  tenantGtid: string;
   230→  deviceFingerprint: string;
   231→  action: string;
   232→  tradeValueUsd?: number;
   233→}): Promise<{ required: boolean; steps: typeof STEP_UP_CHAIN; completed: boolean }> {
   234→  const requiresQes = (params.tradeValueUsd || 0) > 100000;
   235→  const requiresHighValue = (params.tradeValueUsd || 0) > 50000;
   236→  const steps = STEP_UP_CHAIN.filter(s => s.required || (requiresQes && s.requiredFor?.includes("QES")) || (requiresHighValue && s.requiredFor?.includes("high_value")));
   237→  return { required: requiresQes || requiresHighValue, steps, completed: !requiresQes && !requiresHighValue };
   238→}

   240→// 1.14.2 Device Management Center actions
   241→export async function manageDevice(params: {
   242→  deviceFingerprint: string;
   243→  action: "rename" | "revoke" | "force_logout" | "unblock";
   244→  newName?: string;
   245→}): Promise<{ success: boolean }> {
   246→  const device = await db.deviceTrust.findUnique({ where: { deviceFingerprint: params.deviceFingerprint } });
   247→  if (!device) return { success: false };
   248→  switch (params.action) {
   249→    case "rename":
   250→      await db.deviceTrust.update({ where: { deviceFingerprint: params.deviceFingerprint }, data: { deviceName: params.newName || device.deviceName } });
   251→      break;
   252→    case "revoke":
   253→      await db.deviceTrust.update({ where: { deviceFingerprint: params.deviceFingerprint }, data: { state: "REVOKED" } });
   254→      break;
   255→    case "force_logout":
   256→      // In production: invalidate JWT sessions for this device
   257→      await db.sessionAuditEvent.create({ data: { tenantGtid: device.tenantGtid, deviceFingerprint: params.deviceFingerprint, eventType: "logout", description: "Forced logout by admin", ipAddress: device.lastSeenIp } });
   258→      break;
   259→    case "unblock":
   260→      await db.deviceTrust.update({ where: { deviceFingerprint: params.deviceFingerprint }, data: { state: "TRUSTED", riskScore: 0 } });
   261→      break;
   262→  }
   263→  await db.sessionAuditEvent.create({ data: { tenantGtid: device.tenantGtid, deviceFingerprint: params.deviceFingerprint, eventType: "policy_change", description: `Device ${params.action} executed`, ipAddress: device.lastSeenIp } });
   264→  return { success: true };
   265→}

   276→// 1.14.4 Session Risk Engine verdicts
   277→export type SessionRiskVerdict = "ALLOW" | "REQUIRE_REAUTH" | "LOCK_SESSION" | "ESCALATE";
   278→
   279→export async function evaluateSessionRisk(params: {
   280→  tenantGtid: string;
   281→  deviceFingerprint: string;
   282→  ipAddress: string;
   283→  countryCode: string;
   284→}): Promise<{ verdict: SessionRiskVerdict; riskScore: number; reasons: string[] }> {
   285→  const device = await db.deviceTrust.findUnique({ where: { deviceFingerprint: params.deviceFingerprint } });
   286→  let riskScore = 0;
   287→  const reasons: string[] = [];
   288→
   289→  if (device?.state === "BLOCKED" || device?.state === "REVOKED") {
   290→    return { verdict: "LOCK_SESSION", riskScore: 100, reasons: ["Device blocked/revoked"] };
   291→  }
   292→  // Impossible travel (simplified)
   293→  if (device?.lastSeenCountry && device.lastSeenCountry !== params.countryCode) {
   294→    riskScore += 40;
   295→    reasons.push(`Country mismatch: ${device.lastSeenCountry} → ${params.countryCode}`);
   296→  }
   297→  // TOR/VPN detection (simplified)
   298→  if (params.ipAddress.startsWith("10.") || params.ipAddress.startsWith("172.")) {
   299→    riskScore += 20;
   300→    reasons.push("VPN detected");
   301→  }
   302→
   303→  const verdict: SessionRiskVerdict = riskScore >= 70 ? "LOCK_SESSION" : riskScore >= 40 ? "REQUIRE_REAUTH" : riskScore >= 20 ? "ESCALATE" : "ALLOW";
   304→
   305→  if (reasons.length > 0) {
   306→    await db.sessionRiskEvent.create({
   307→      data: {
   308→        tenantGtid: params.tenantGtid, deviceFingerprint: params.deviceFingerprint,
   309→        eventType: reasons[0].split(":")[0].toLowerCase().replace(/ /g, "_") as any,
   310→        severity: riskScore >= 70 ? "critical" : riskScore >= 40 ? "high" : "medium",
   311→        description: reasons.join("; "), ipAddress: params.ipAddress, countryCode: params.countryCode,
   312→      },
   313→    });
   314→  }
   315→  return { verdict, riskScore, reasons };
   316→}

   743→export const SCREENING_DIMENSIONS = [

   750→
   751→export async function runComplianceScreening(params: {
   752→  tenantGtid: string;
   753→  ustn?: string;
   754→  counterpartyGtid?: string;
   755→}): Promise<{ overall: ScreeningVerdict; results: { dimension: string; verdict: ScreeningVerdict; details: string }[] }> {
   756→  const tenant = await db.tenant.findUnique({ where: { gtid: params.tenantGtid } });
   757→  const counterparty = params.counterpartyGtid ? await db.tenant.findUnique({ where: { gtid: params.counterpartyGtid } }) : null;
   758→  const results: { dimension: string; verdict: ScreeningVerdict; details: string }[] = [];
   759→
   760→  // 1. Sanctions (simulated — in production: RIA + GNN)
   761→  const sanctionsClear = tenant?.sanctionsCleared && (!counterparty || counterparty.sanctionsCleared);
   762→  results.push({
   763→    dimension: "SANCTIONS",
   764→    verdict: sanctionsClear ? "CLEAR" : "BLOCKED",
   765→    details: sanctionsClear ? "No sanctions hits (UN, OFAC, EU lists). GNN proximity > 2 hops." : "Sanctions hit detected — trade blocked.",
   766→  });
   767→
   768→  // 2. PEP (simulated)
   769→  const isPep = tenant?.kybTier === 1; // lower tier = higher PEP risk
   770→  results.push({
   771→    dimension: "PEP",
   772→    verdict: isPep ? "ENHANCED_DUE_DILIGENCE" : "CLEAR",
   773→    details: isPep ? "UBO flagged as Politically Exposed Person — enhanced DD required." : "No PEP matches in global databases.",
   774→  });
   775→
   776→  // 3. Restricted goods (based on HS code)
   777→  const restrictedHs = ["8802", "9301", "9303", "3601"]; // aircraft, weapons, explosives
   778→  const trade = params.ustn ? await db.trade.findUnique({ where: { ustn: params.ustn } }) : null;
   779→  const hsFlagged = trade?.commodityHs && restrictedHs.some(hs => trade.commodityHs.startsWith(hs));
   780→  results.push({
   781→    dimension: "RESTRICTED_GOODS",
   782→    verdict: hsFlagged ? "BLOCKED" : "CLEAR",
   783→    details: hsFlagged ? `HS code ${trade?.commodityHs} flagged as dual-use/restricted goods.` : "No restricted goods detected (dual-use, CITES, military).",
   784→  });
   785→
   786→  // 4. Jurisdiction risk
   787→  const buyerJur = trade ? await db.jurisdiction.findUnique({ where: { countryCode: trade.buyerGtid.slice(5, 7) } }) : null;
   788→  const sellerJur = tenant ? await db.jurisdiction.findUnique({ where: { countryCode: tenant.country } }) : null;
   789→  const strictest = buyerJur?.tier === "BLOCKED" || sellerJur?.tier === "BLOCKED" ? "BLOCKED"
   790→    : buyerJur?.tier === "RESTRICTED" || sellerJur?.tier === "RESTRICTED" ? "RESTRICTED"
   791→    : buyerJur?.tier === "LIMITED" || sellerJur?.tier === "LIMITED" ? "LIMITED" : "FULL";
   792→  results.push({
   793→    dimension: "JURISDICTION_RISK",
   794→    verdict: strictest === "BLOCKED" ? "BLOCKED" : strictest === "RESTRICTED" || strictest === "LIMITED" ? "ENHANCED_DUE_DILIGENCE" : "CLEAR",
   795→    details: `Strictest jurisdiction tier: ${strictest} (buyer: ${buyerJur?.tier || "?"}, seller: ${sellerJur?.tier || "?"}).`,
   796→  });
   797→
   798→  // 5. Customs compliance (simulated — based on declaration history)
   799→  const customsOk = tenant?.lifecycleState === "VERIFIED";
   800→  results.push({
   801→    dimension: "CUSTOMS_COMPLIANCE",
   802→    verdict: customsOk ? "CLEAR" : "ENHANCED_DUE_DILIGENCE",
   803→    details: customsOk ? "Tenant VERIFIED — customs compliance history clean." : "Tenant not fully verified — customs review required.",
   804→  });
   805→
   806→  // Persist results
   807→  for (const r of results) {
   808→    await db.complianceScreening.create({
   809→      data: {
   810→        ustn: params.ustn || null,
   811→        tenantGtid: params.tenantGtid,
   812→        screeningType: r.dimension,
   813→        verdict: r.verdict,
   814→        dataSource: SCREENING_DIMENSIONS.find(d => d.id === r.dimension)?.source || "RIA",
   815→        details: r.details,
   816→      },
   817→    });
   818→  }
   819→
   820→  const overall: ScreeningVerdict = results.some(r => r.verdict === "BLOCKED") ? "BLOCKED"
   821→    : results.some(r => r.verdict === "ENHANCED_DUE_DILIGENCE") ? "ENHANCED_DUE_DILIGENCE" : "CLEAR";
   822→
   823→  return { overall, results };
   824→}

   836→// ============ Part 1.16.4: Compliance Override & Appeal (multisig) ============
   837→export async function overrideComplianceVerdict(params: {
   838→  screeningId: string;
   839→  reason: string;
   840→  approverGtids: string[]; // multisig approvers
   841→}): Promise<{ success: boolean; requiredApprovals: number; currentApprovals: number }> {
   842→  const screening = await db.complianceScreening.findUnique({ where: { id: params.screeningId } });
   843→  if (!screening) return { success: false, requiredApprovals: 0, currentApprovals: 0 };
   844→
   845→  // 3/5 multisig for BLOCKED, 1/3 for ENHANCED_DUE_DILIGENCE
   846→  const requiredApprovals = screening.verdict === "BLOCKED" ? 3 : 1;
   847→  const currentApprovals = params.approverGtids.length;
   848→
   849→  if (currentApprovals < requiredApprovals) {
   850→    return { success: false, requiredApprovals, currentApprovals };
   851→  }
   852→
   853→  await db.complianceScreening.update({
   854→    where: { id: params.screeningId },
   855→    data: {
   856→      overridden: true,
   857→      overrideReason: params.reason,
   858→      overrideMultisig: JSON.stringify(params.approverGtids),
   859→      reviewed: true,
   860→      reviewedBy: params.approverGtids[0],
   861→    },
   862→  });
   863→
   864→  // Log as Governor decision
   865→  await db.governorDecision.create({
   866→    data: {
   867→      decisionId: "dec-override-" + Date.now().toString(36),
   868→      action: "compliance_override",
   869→      actorGtid: params.approverGtids[0],
   870→      verdict: "ALLOW",
   871→      conditions: JSON.stringify([{ condition_id: "override", label: `Override of ${screening.verdict} verdict: ${params.reason}`, status: "met" }]),
   872→      loomHash: "sha256:" + createHash("sha256").update(params.screeningId + params.reason + Date.now()).digest("hex"),
   873→      previousHash: null,
   874→      signature: "ed25519:" + createHash("sha256").update(params.screeningId + "::sgtx-platform-key").digest("hex").slice(0, 64),
   875→      moduleVersions: "{}",
   876→    },
   877→  });
   878→
   879→  return { success: true, requiredApprovals, currentApprovals };
   880→}
