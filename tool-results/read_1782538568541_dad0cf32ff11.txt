   200→  // must designate a licensed customs broker for their respective side of
   201→  // clearance (seller = export, buyer = import). They may designate their

   203→  // OR a dedicated customs broker (CBR tenant). The designated broker

   228→  buyerAddress    String?
   229→  buyerTaxId      String?

   231→  consigneeSameAsBuyer Boolean @default(false)

   240→  updatedAt     DateTime @updatedAt

   243→  @@index([ustn])

   261→  arbitrationClause String   // ICC | LCIA | CRCICA | DIFC | UNCITRAL
   262→  arbitrationSeat   String   // Cairo | London | Paris | Dubai | Singapore
   263→  language          String   @default("en") // en | ar | de | fr | dual
