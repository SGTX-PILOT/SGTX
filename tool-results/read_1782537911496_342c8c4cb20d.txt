    22→const INFERENCE_LOG: InferenceRecord[] = [];
    23→const MAX_LOG = 200;
    24→
    25→function logInference(rec: Omit<InferenceRecord, "created_at">) {
    26→  INFERENCE_LOG.push({ ...rec, created_at: new Date().toISOString() });
    27→  if (INFERENCE_LOG.length > MAX_LOG) INFERENCE_LOG.shift();
    28→}

    30→export function getInferenceLog(limit = 50): InferenceRecord[] {
    31→  return INFERENCE_LOG.slice(-limit).reverse();

    44→async function callZAI(systemPrompt: string, userPrompt: string, opts: { maxTokens?: number; temperature?: number } = {}): Promise<{ content: string; latencyMs: number }> {
    45→  const zai = await getZAI();
    46→  const start = Date.now();
    47→  const completion = await zai.chat.completions.create({
    48→    messages: [
    49→      { role: "system", content: systemPrompt },
    50→      { role: "user", content: userPrompt },
    51→    ],
    52→    thinking: { type: "disabled" },
    53→    max_tokens: opts.maxTokens ?? 400,
    54→    temperature: opts.temperature ?? 0.4,
    55→  });
    56→  const latencyMs = Date.now() - start;
    57→  const content = completion.choices?.[0]?.message?.content || "";
    58→  return { content, latencyMs };
    59→}

    61→// ============ HuggingFace (A2/A3 secondary) ============
    62→async function callHuggingFace(modelId: string, systemPrompt: string, userPrompt: string, opts: { maxTokens?: number; temperature?: number } = {}): Promise<{ content: string; latencyMs: number }> {
    63→  const apiKey = process.env.HF_API_TOKEN;
    64→  if (!apiKey) throw new Error("HF_API_TOKEN not configured");
    65→
    66→  const url = `https://api-inference.huggingface.co/models/${modelId}`;
    67→  const start = Date.now();
    68→  const payload = {
    69→    inputs: `<s>[INST] ${systemPrompt}\n\n${userPrompt} [/INST]`,
    70→    parameters: {
    71→      max_new_tokens: opts.maxTokens ?? 400,
    72→      temperature: opts.temperature ?? 0.4,
    73→      return_full_text: false,
    74→    },
    75→    options: { wait_for_model: true },
    76→  };
    77→  const res = await fetch(url, {
    78→    method: "POST",
    79→    headers: { Authorization: `Bearer ${apiKey}`, "Content-Type": "application/json" },
    80→    body: JSON.stringify(payload),
    81→  });
    82→  const latencyMs = Date.now() - start;
    83→  if (!res.ok) {
    84→    const errText = await res.text().catch(() => "");
    85→    throw new Error(`HF ${res.status}: ${errText.slice(0, 200)}`);
    86→  }
    87→  const data = await res.json();
    88→  const content = Array.isArray(data) ? data[0]?.generated_text || "" : data.generated_text || "";
    89→  return { content: content.trim(), latencyMs };
    90→}

   112→export interface AIResult {
   113→  content: string;
   114→  provider: AIProvider;
   115→  model: string;
   116→  latencyMs: number;
   117→  fallbackUsed: boolean;
   118→  authority: AuthorityLevel;

   121→export async function runAI(params: {
   122→  agentName: string;
   123→  authority: AuthorityLevel;
   124→  systemPrompt: string;
   125→  userPrompt: string;
   126→  fallbackKey: string;
   127→  maxTokens?: number;
   128→  temperature?: number;
   129→}): Promise<AIResult> {
   130→  const { agentName, authority, systemPrompt, userPrompt, fallbackKey, maxTokens, temperature } = params;
   131→
   132→  // A0 (Observational): Logging only, no influence. Returns empty content, logs the observation.
   133→  if (authority === "A0") {
   134→    logInference({ agent_name: agentName, authority_level: "A0", provider: "static", model: "observational", latency_ms: 0, fallback_used: false, output_length_tokens: 0, input_context: userPrompt.slice(0, 200), success: true });
   135→    return { content: "", provider: "static", model: "observational", latencyMs: 0, fallbackUsed: false, authority: "A0" };
   136→  }
   137→
   138→  // A4 (Governance): Auto-executes within constitutional bounds via OPA + WasmEdge.
   139→  // Returns a deterministic governance decision (not an AI inference — it's a rule execution).
   140→  if (authority === "A4") {
   141→    logInference({ agent_name: agentName, authority_level: "A4", provider: "opa_wasm", model: "constitutional-rules", latency_ms: 1, fallback_used: false, output_length_tokens: 0, input_context: userPrompt.slice(0, 200), success: true });
   142→    return { content: "ALLOW: Constitutional bounds satisfied. Auto-execution permitted.", provider: "opa_wasm", model: "constitutional-rules", latencyMs: 1, fallbackUsed: false, authority: "A4" };
   143→  }
   144→
   145→  // A5 (FORBIDDEN): No autonomous execution. Blocked at compile time (simulated).
   146→  if (authority === "A5") {
   147→    logInference({ agent_name: agentName, authority_level: "A5", provider: "blocked", model: "forbidden", latency_ms: 0, fallback_used: false, output_length_tokens: 0, input_context: "A5_ATTEMPT_BLOCKED", success: false, error: "Constitutional violation: A5 autonomous execution attempt blocked at WASM compile time." });
   148→    return { content: "BLOCKED: A5 autonomous execution is forbidden by constitutional law.", provider: "blocked", model: "forbidden", latencyMs: 0, fallbackUsed: false, authority: "A5" };
   149→  }
   150→
   151→  // A1: z-ai primary → static
   152→  if (authority === "A1") {
   153→    try {
   154→      const { content, latencyMs } = await callZAI(systemPrompt, userPrompt, { maxTokens, temperature });
   155→      logInference({ agent_name: agentName, authority_level: authority, provider: "zai", model: "glm-4-plus", latency_ms: latencyMs, fallback_used: false, output_length_tokens: Math.ceil(content.length / 4), input_context: userPrompt.slice(0, 200), success: true });
   156→      return { content, provider: "zai", model: "glm-4-plus", latencyMs, fallbackUsed: false, authority };
   157→    } catch (err: any) {
   158→      logInference({ agent_name: agentName, authority_level: authority, provider: "zai", model: "glm-4-plus", latency_ms: 0, fallback_used: true, output_length_tokens: 0, input_context: userPrompt.slice(0, 200), success: false, error: err.message });
   159→      return { content: staticFallback(fallbackKey), provider: "static", model: "static-template", latencyMs: 0, fallbackUsed: true, authority };
   160→    }
   161→  }
   162→
   163→  // A2/A3: z-ai primary → HuggingFace secondary → static
   164→  try {
   165→    const { content, latencyMs } = await callZAI(systemPrompt, userPrompt, { maxTokens, temperature });
   166→    logInference({ agent_name: agentName, authority_level: authority, provider: "zai", model: "glm-4-plus", latency_ms: latencyMs, fallback_used: false, output_length_tokens: Math.ceil(content.length / 4), input_context: userPrompt.slice(0, 200), success: true });
   167→    return { content, provider: "zai", model: "glm-4-plus", latencyMs, fallbackUsed: false, authority };
   168→  } catch (err: any) {
   169→    logInference({ agent_name: agentName, authority_level: authority, provider: "zai", model: "glm-4-plus", latency_ms: 0, fallback_used: true, output_length_tokens: 0, input_context: userPrompt.slice(0, 200), success: false, error: err.message });
   170→    // Fallback: HuggingFace Mixtral
   171→    const hfModel = process.env.HF_MIXTRAL_MODEL || "mistralai/Mistral-7B-Instruct-v0.3";
   172→    try {
   173→      const { content, latencyMs } = await callHuggingFace(hfModel, systemPrompt, userPrompt, { maxTokens, temperature });
   174→      logInference({ agent_name: agentName, authority_level: authority, provider: "huggingface", model: hfModel, latency_ms: latencyMs, fallback_used: true, output_length_tokens: Math.ceil(content.length / 4), input_context: userPrompt.slice(0, 200), success: true });
   175→      return { content, provider: "huggingface", model: hfModel, latencyMs, fallbackUsed: true, authority };
   176→    } catch (err2: any) {
   177→      logInference({ agent_name: agentName, authority_level: authority, provider: "huggingface", model: hfModel, latency_ms: 0, fallback_used: true, output_length_tokens: 0, input_context: userPrompt.slice(0, 200), success: false, error: err2.message });
   178→      return { content: staticFallback(fallbackKey), provider: "static", model: "static-template", latencyMs: 0, fallbackUsed: true, authority };
   179→    }
   180→  }
   181→}

   269→/** A2 — Governor pre-screen (Part 1.4, compliance prescreen) */
   270→export async function governorPrescreen(trade: {
   271→  commodity: string;
   272→  hsCode: string;
   273→  buyerCountry: string;
   274→  sellerCountry: string;
   275→  value: number;
   276→  // Part 4.15 expanded fields — all optional for backwards compat
   277→  incoterm?: string;
   278→  transportMode?: string;
   279→  equipmentType?: string;
   280→  insuranceRequirement?: string;
   281→  insuranceType?: string;
   282→  settlementStructure?: string;
   283→  paymentTiming?: string;
   284→  currency?: string;
   285→  tradeCriticality?: string;
   286→  earliestDeliveryDate?: string;
   287→  preferredDeliveryDate?: string;
   288→  latestDeliveryDate?: string;
   289→  documentationMandatoryCount?: number;
   290→  documentationMandatorySelected?: number;
   291→  sellerGtid?: string;
   292→}): Promise<AIResult & { verdict: string; conditions: string[] }> {
   293→  // ── Heuristic rule-based checks (A4 Governor gates) — run before AI for fast fail ──
   294→  const heuristicConditions: string[] = [];
   295→  let heuristicVerdict: "ALLOW" | "CONDITIONAL" | "DENY" = "ALLOW";
   296→
   297→  // G1U18 — Transport mode valid
   298→  if (trade.transportMode === undefined) {
   299→    // not provided → skip check (caller may not have set it yet)
   300→  } else if (trade.transportMode && !["OCEAN", "AIR", "RAIL", "TRUCK", "MULTIMODAL"].includes(trade.transportMode)) {
   301→    heuristicConditions.push("Transport mode invalid (G1U18)");
   302→    heuristicVerdict = "CONDITIONAL";
   303→  }
   304→
   305→  // G1U19 — Equipment compatible with transport mode
   306→  if (trade.transportMode && trade.equipmentType) {
   307→    const validOcean = ["STANDARD", "HC", "REEFER", "OPEN_TOP", "FLAT_RACK", "TANK"];
   308→    const validAir = ["ULD_PALLET", "ULD_CONTAINER", "BULK"];
   309→    const validTruck = ["DRY_VAN", "REEFER_TRUCK", "FLATBED", "CURTAIN_SIDE"];
   310→    const validRail = ["BOX_WAGON", "FLAT_WAGON", "TANK_WAGON", "REEFER_WAGON"];
   311→    const valid = trade.transportMode === "OCEAN" ? validOcean
   312→      : trade.transportMode === "AIR" ? validAir
   313→      : trade.transportMode === "TRUCK" ? validTruck
   314→      : trade.transportMode === "RAIL" ? validRail
   315→      : [...validOcean, ...validAir, ...validTruck, ...validRail];
   316→    if (!valid.includes(trade.equipmentType)) {
   317→      heuristicConditions.push(`Equipment ${trade.equipmentType} not compatible with ${trade.transportMode} (G1U19)`);
   318→      heuristicVerdict = "CONDITIONAL";
   319→    }
   320→  }
   321→
   322→  // G1U20a/d — Insurance requirement selected + CIF/CIP requires insurance
   323→  if (trade.incoterm && ["CIF", "CIP"].includes(trade.incoterm)) {
   324→    if (!trade.insuranceRequirement || trade.insuranceRequirement !== "REQUIRED") {
   325→      heuristicConditions.push(`Incoterm ${trade.incoterm} requires insurance to be REQUIRED (G1U20d)`);
   326→      heuristicVerdict = "CONDITIONAL";
   327→    }
   328→  }
   329→
   330→  // G1U9/G1U10/G1U12 — Commercial settlement validity
   331→  if (trade.settlementStructure !== undefined && !trade.settlementStructure) {
   332→    heuristicConditions.push("Settlement structure not selected (G1U9)");
   333→    heuristicVerdict = "CONDITIONAL";
   334→  }
   335→  if (trade.paymentTiming !== undefined && !trade.paymentTiming) {
   336→    heuristicConditions.push("Payment timing not selected (G1U10)");
   337→    heuristicVerdict = "CONDITIONAL";
   338→  }
   339→
   340→  // G1U21 — Required documents selected
   341→  if (trade.documentationMandatoryCount !== undefined && trade.documentationMandatorySelected !== undefined) {
   342→    if (trade.documentationMandatoryCount > 0 && trade.documentationMandatorySelected < trade.documentationMandatoryCount) {
   343→      heuristicConditions.push(`Mandatory documents incomplete: ${trade.documentationMandatorySelected}/${trade.documentationMandatoryCount} selected (G1U21)`);
   344→      heuristicVerdict = "CONDITIONAL";
   345→    }
   346→  }
   347→
   348→  // Delivery window validation (G1U20)
   349→  if (trade.earliestDeliveryDate && trade.preferredDeliveryDate && trade.latestDeliveryDate) {
   350→    const e = new Date(trade.earliestDeliveryDate).getTime();
   351→    const p = new Date(trade.preferredDeliveryDate).getTime();
   352→    const l = new Date(trade.latestDeliveryDate).getTime();
   353→    const now = Date.now();
   354→    if (e <= now || p <= now || l <= now) {
   355→      heuristicConditions.push("Delivery window contains past dates (G1U20)");
   356→      heuristicVerdict = "CONDITIONAL";
   357→    }
   358→    if (!(e <= p && p <= l)) {
   359→      heuristicConditions.push("Delivery window order invalid (earliest ≤ preferred ≤ latest) (G1U20)");
   360→      heuristicVerdict = "CONDITIONAL";
   361→    }
   362→    const windowDays = (l - e) / (1000 * 60 * 60 * 24);
   363→    if (windowDays > 60) {
   364→      heuristicConditions.push(`Delivery window too large (${Math.round(windowDays)} days > 60 max) (G1U20)`);
   365→      heuristicVerdict = "CONDITIONAL";
   366→    }
   367→  }
   368→
   369→  // Criticality appropriate for incoterm (G1U11b)
   370→  if (trade.tradeCriticality === "CRITICAL" && trade.incoterm && ["EXW", "FOB"].includes(trade.incoterm)) {
   371→    heuristicConditions.push("Critical criticality not recommended for EXW/FOB incoterms (G1U11b)");
   372→    heuristicVerdict = "CONDITIONAL";
   373→  }
   374→
   375→  // ── AI advisory (A2) — runs in parallel for risk evaluation ──
   376→  const result = await runAI({
   377→    agentName: "governor_prescreen",
   378→    authority: "A2",
   379→    systemPrompt: 'You are the SGTX Governor Pre-Screen AI (A2 constraining). Evaluate the trade for compliance risks.    388→    const match = result.content.match(/\{[\s\S]*\}/);
   389→    if (match) {

   391→      verdict = parsed.verdict || "ALLOW";

   394→  } catch {}
   395→
   396→  // Merge heuristic + AI: DENY > CONDITIONAL > ALLOW
   397→  const merged: Record<string, number> = { ALLOW: 0, CONDITIONAL: 1, DENY: 2 };

   399→  const mergedConditions = Array.from(new Set([...heuristicConditions, ...conditions]));
   400→  return { ...result, verdict: finalVerdict, conditions: mergedConditions };
