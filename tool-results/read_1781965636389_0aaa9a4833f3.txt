   158→const useStagger = (stagger = 0.08, delay = 0): Variants => {
   159→  const reduce = useReducedMotion();
   160→  return useMemo(
   161→    () => ({
   162→      hidden: { opacity: 0 },
   163→      show: {
   164→        opacity: 1,
   165→        transition: { staggerChildren: reduce ? 0 : stagger, delayChildren: delay },
   166→      },
   167→    }),
   168→    [reduce, stagger, delay],
   169→  );
   170→};

   231→function ParticleField() {
   232→  const reduce = useReducedMotion();
   233→  const particles = useMemo(
   234→    () =>
   235→      Array.from({ length: 28 }).map((_, i) => ({
   236→        id: i,
   237→        left: (i * 37) % 100,
   238→        top: (i * 53) % 100,
   239→        size: 1 + (i % 4),
   240→        delay: (i % 7) * 0.6,
   241→        duration: 6 + (i % 6),
   242→        gold: i % 3 === 0,
   243→      })),
   244→    [],
   245→  );
   246→  if (reduce) return null;
   247→  return (
   248→    <div className="absolute inset-0 pointer-events-none overflow-hidden">
   249→      {particles.map((p) => (
   250→        <motion.span
   251→          key={p.id}
   252→          className="absolute rounded-full"
   253→          style={{
   254→            left: `${p.left}%`,
   255→            top: `${p.top}%`,
   256→            width: p.size,
   257→            height: p.size,
   258→            background: p.gold ? "oklch(0.84 0.14 84)" : "oklch(0.92 0.01 250)",
   259→            boxShadow: p.gold ? "0 0 6px oklch(0.84 0.14 84 / 0.6)" : "0 0 4px oklch(0.92 0.01 250 / 0.4)",
   260→          }}
   261→          animate={{ opacity: [0, 0.8, 0], scale: [0.5, 1.3, 0.5], y: [0, -22, 0] }}
   262→          transition={{ duration: p.duration, repeat: Infinity, delay: p.delay, ease: "easeInOut" }}
   263→        />
   264→      ))}
   265→    </div>
   266→  );
   267→}
