   291→// ============ Network / Saved Contacts (Part 2.6) ============
   292→export function NetworkScreen({ tenantGtid }: { tenantGtid: string }) {
   293→  const qc = useQueryClient();
   294→  const [addGtid, setAddGtid] = useState("");
   295→  const [adding, setAdding] = useState(false);
   296→  const { data: contacts, isLoading } = useQuery({
   297→    queryKey: ["contacts", tenantGtid],
   298→    queryFn: async () => (await fetch(`/api/sgtx/contacts?owner=${tenantGtid}`)).json(),
   299→  });
   300→
   301→  const add = async () => {
   302→    if (!addGtid || adding) return;
   303→    setAdding(true);
   304→    try {
   305→      await fetch("/api/sgtx/contacts", {
   306→        method: "POST", headers: { "Content-Type": "application/json" },
   307→        body: JSON.stringify({ ownerGtid: tenantGtid, contactGtid: addGtid }),
   308→      });
   309→      setAddGtid("");
   310→      qc.invalidateQueries({ queryKey: ["contacts", tenantGtid] });
   311→    } catch {}
   312→    finally { setAdding(false); }
   313→  };
   314→
   315→  return (
   316→    <div className="space-y-4">
   317→      <SectionHeader title="Network — Saved Contacts" subtitle="Part 2.6 — opt-in directory of existing counterparties · AI Trust Portrait · no marketplace discovery" />

   319→        <div className="flex items-center gap-2">

   321→          <Button onClick={add} disabled={adding || !addGtid} className="bg-gold-gradient text-sovereign h-9">
   322→            {adding ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Users className="w-3.5 h-3.5 mr-1.5" />} Add Contact
   323→          </Button>
   324→        </div>
   325→        <p className="text-[0.6rem] text-muted-foreground mt-2">🔐 You must already know the GTID. SGTX never recommends "people you may know". AI resolves + generates a Trust Portrait.</p>
   326→      </Card>

   328→        {(contacts || []).map((c: any) => (
   329→          <Card key={c.id} className="p-4 hover:border-gold/40 transition-colors">
   330→            <div className="flex items-start justify-between mb-2">
   331→              <div className="flex items-center gap-2">

   333→                <div>
   334→                  <p className="text-sm font-semibold">{c.contactName}</p>

   336→                </div>
   337→              </div>
   338→              <Badge variant="outline" className="text-[0.6rem]">{c.contactType}</Badge>
   339→            </div>
   340→            <div className="flex items-center gap-3 text-xs mb-2">
   341→              <span className="flex items-center gap-1"><ShieldCheck className="w-3 h-3" style={{ color: healthColor(c.healthScore) }} /> {c.healthScore}</span>
   342→              <span className="text-muted-foreground">{c.totalTrades} trades</span>
   343→              <Badge variant="outline" className="text-[0.55rem]">{c.relationship}</Badge>
