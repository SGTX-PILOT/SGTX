   288→function InboxDrawer({ data, onClose, highPriority }: { data: DashboardData; onClose: () => void; highPriority: number }) {
   289→  const [aiSummary, setAiSummary] = useState<string | null>(null);
   290→  const [aiLoading, setAiLoading] = useState(false);
   291→  const [aiProvider, setAiProvider] = useState<string | null>(null);
   292→
   293→  const loadSummary = async () => {
   294→    if (aiLoading || aiSummary) return;
   295→    setAiLoading(true);
   296→    try {
   297→      const res = await fetch("/api/sgtx/ai/inbox-summary", {
   298→        method: "POST", headers: { "Content-Type": "application/json" },
   299→        body: JSON.stringify({ tenant: data.tenant.gtid }),
   300→      });
   301→      const d = await res.json();
   302→      setAiSummary(d.content);
   303→      setAiProvider(d.provider);
   304→    } catch { setAiSummary("AI summary unavailable."); }
   305→    finally { setAiLoading(false); }
   306→  };
   307→
   308→  return (
   309→    <>
   310→      <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} onClick={onClose} className="fixed inset-0 bg-black/50 z-40" />
   311→      <motion.div
   312→        initial={{ x: "100%" }} animate={{ x: 0 }} exit={{ x: "100%" }} transition={{ type: "spring", damping: 28 }}

   315→        <div className="h-16 flex items-center justify-between px-5 border-b border-border">

   317→            <h3 className="font-semibold text-sm">Smart Inbox</h3>

   386→function AssistantDrawer({ onClose, tenant }: { onClose: () => void; tenant: any }) {
   387→  const [messages, setMessages] = useState<{ role: "user" | "ai"; content: string; provider?: string }[]>(
   388→    tenant ? [{ role: "ai", content: `Hello, ${tenant.legalName}. I'm your SGTX AI Operations Assistant & Customer Care Chatbot (A1 advisory, z-ai glm-4-plus). I can answer questions, perform actions on your behalf (with PIN-based impersonation), and escalate to human support via VoIP callback. I never recommend counterparties — SGTX is a non-marketplace system.` }] : []
   389→  );
   390→  const [input, setInput] = useState("");
   391→  const [loading, setLoading] = useState(false);
   392→  const [showEscalate, setShowEscalate] = useState(false);
   393→  const [pinPrompt, setPinPrompt] = useState<string | null>(null);
   394→  const [pinInput, setPinInput] = useState("");
   395→  const scrollRef = useRef<HTMLDivElement>(null);
   396→
   397→  const send = async (text?: string) => {
   398→    const msg = (text ?? input).trim();
   399→    if (!msg || loading || !tenant) return;
   400→    setInput("");
   401→
   402→    // Handle "Talk to a human" escalation (3B.1.3.6)

   404→      setMessages((m) => [...m, { role: "user", content: msg }, { role: "ai", content: "I'll escalate this to a human support agent.
