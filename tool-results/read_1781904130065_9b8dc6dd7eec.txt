  1022→-- Tenants RLS
  1023→
  1024→CREATE POLICY tenants_tenant_isolation ON tenants
  1025→
  1026→USING (gtid = current_setting('app.current_tenant_gtid')::text OR
  1027→
  1028→EXISTS (SELECT 1 FROM employees WHERE tenant_id = id AND employees.id =
  1029→current_setting('app.current_employee_id')::uuid));
  1030→
  1031→-- Employees RLS
  1032→
  1033→CREATE POLICY employees_tenant_isolation ON employees
  1034→
  1035→USING (tenant_id = current_setting('app.current_tenant_id')::uuid);
  1036→
  1037→-- Trade requests RLS
  1038→
  1039→CREATE POLICY trade_requests_tenant_isolation ON trade_requests
  1040→
  1041→USING (buyer_tenant_id = current_setting('app.current_tenant_id')::uuid
  1042→OR
  1043→
  1044→seller_tenant_id = current_setting('app.current_tenant_id')::uuid);
  1045→
  1046→-- Shipments RLS
  1047→
  1048→CREATE POLICY shipments_tenant_isolation ON shipments
  1049→
  1050→USING (exporter_gtid = current_setting('app.current_tenant_gtid')::text
  1051→OR
  1052→
  1053→importer_gtid = current_setting('app.current_tenant_gtid')::text OR
  1054→
  1055→EXISTS (SELECT 1 FROM service_quotations WHERE ustn = shipments.ustn AND
  1056→provider_gtid = current_setting('app.current_tenant_gtid')::text));
  1057→
  1058→-- Documents RLS
  1059→
  1060→CREATE POLICY documents_tenant_isolation ON documents
  1061→
  1062→USING (tenant_id = current_setting('app.current_tenant_id')::uuid OR
  1063→
  1064→EXISTS (SELECT 1 FROM shipments WHERE ustn = documents.ustn AND
  1065→
  1066→(exporter_gtid = current_setting('app.current_tenant_gtid')::text OR
  1067→
  1068→importer_gtid = current_setting('app.current_tenant_gtid')::text)));
  1069→
  1070→-- Disputes RLS

  1072→CREATE POLICY disputes_tenant_isolation ON disputes
  1073→
  1074→USING (filing_party_gtid =
  1075→current_setting('app.current_tenant_gtid')::text OR
  1076→
  1077→EXISTS (SELECT 1 FROM shipments WHERE ustn = disputes.ustn AND
  1078→
  1079→(exporter_gtid = current_setting('app.current_tenant_gtid')::text OR

  1081→importer_gtid = current_setting('app.current_tenant_gtid')::text)));
